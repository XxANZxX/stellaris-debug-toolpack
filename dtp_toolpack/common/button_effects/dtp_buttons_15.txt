					if = {
						limit = {
							has_star_flag = dtp_marked
						}
						dtp_unmark_sys = yes
					}
					else = {
						dtp_mark_sys = yes
					}
				}
				dtp_result = { CODE = 36 }
			}
			else = {
				dtp_deny = { CODE = 36 }
			}

			solar_system = {
				log = "DTP STRIP system=[This.GetName]"

					limit = {
						has_star_flag = dtp_marked
					}
					log = "DTP STRIP system now MARKED"
				}

					log = "DTP STRIP system NOT marked"
