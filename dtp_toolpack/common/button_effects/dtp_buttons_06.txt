		from = {
			has_country_flag = dtp_open

		NOT = { dtp_p_hasmod = { M = shooting_stars } }
		from = {
			check_variable = { which = dtp_modpage value = 2 }
		}
	}
	allow = {
		custom_tooltip = { fail_text = dtp_need_perm_planet from = {
				dtp_has_perm = { P = planet }
			} }
		custom_tooltip = { fail_text = dtp_need_tp dtp_has_tp = yes }
	}
	effect = {
		hidden_effect = {
			hidden_effect = {
				from = {
					dtp_pidx_ensure = yes
					dtp_init_targets = yes
				}
				dtp_load = yes
				dtp_disarm = yes
			}
			if = {
				limit = {
					from = {
						dtp_has_perm = { P = planet }
					}
				}
				if = {
					limit = {
						exists = event_target:dtp_p
					}
					hidden_effect = {
						event_target:dtp_p = {
							set_variable = { which = dtp_ty value = from.dtp_years }
							remove_modifier = shooting_stars
						}
						if = {
							limit = {
								from = {
									check_variable = { which = dtp_years value <= 0 }
								}
							}
							event_target:dtp_p = {
								add_modifier = { modifier = shooting_stars days = -1 }
							}
						}
						else = {
							event_target:dtp_p = {
								add_modifier = { modifier = shooting_stars days = 360 time_multiplier = dtp_ty }
							}
						}
					}
					dtp_result = { CODE = 5 }
				}
				else = {
					dtp_fail = { CODE = 5 }

			}
			else = {

			}
		}

}
