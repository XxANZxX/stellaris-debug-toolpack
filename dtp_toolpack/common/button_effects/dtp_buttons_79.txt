						dtp_has_perm = { P = planet }
					}
				}
				if = {
					limit = {
						exists = event_target:dtp_p
						exists = event_target:dtp_t
					}
					event_target:dtp_p = {
						if = {
							limit = {
								is_preventing_anomaly = yes
							}
							prevent_anomaly = no
						}
						add_anomaly = { category = moon_cat target = event_target:dtp_t }
					}
					dtp_result = { CODE = 69 }
				}
				else = {
					dtp_fail = { CODE = 69 }
				}
			}
			else = {
				dtp_deny = { CODE = 69 }
			}
		}
	}
