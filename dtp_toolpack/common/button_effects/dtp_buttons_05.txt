dtp_b_um_c_storage = {
	potential = {
		from = {
			has_country_flag = dtp_open
		}
		from = {
			check_variable = { which = dtp_tab value = 4 }
		}
		NOT = { dtp_c_hasmod = { M = dtp_uc_storage } }
		from = {
			NOT = { check_variable = { which = dtp_modpage value = 2 } }
		}
	}
	allow = {
		custom_tooltip = { fail_text = dtp_need_perm_res from = {
				dtp_has_perm = { P = res }
			} }
		custom_tooltip = { fail_text = dtp_need_tc dtp_has_tc = yes }
	}
	effect = {
		hidden_effect = {
			hidden_effect = {
				from = {
					dtp_pidx_ensure = yes
					dtp_init_targets = yes
				}
				dtp_load = yes
				dtp_disarm = yes
			}
			if = {
				limit = {
					from = {
						dtp_has_perm = { P = res }
					}
				}
				if = {
					limit = {
						exists = event_target:dtp_t
					}
					hidden_effect = {
						event_target:dtp_t = {
							set_variable = { which = dtp_tv value = from.dtp_val }
							set_variable = { which = dtp_ty value = from.dtp_years }
							remove_modifier = dtp_uc_storage
						}
						if = {
							limit = {
								from = {
									check_variable = { which = dtp_years value <= 0 }
								}
							}
							event_target:dtp_t = {
								add_modifier = { modifier = dtp_uc_storage days = -1 multiplier = dtp_tv }
							}
						}
						else = {
							event_target:dtp_t = {
								add_modifier = { modifier = dtp_uc_storage days = 360 time_multiplier = dtp_ty multiplier = dtp_tv }

						}

				}
				else = {
					dtp_fail = { CODE = 5 }
				}
			}
			else = {
				dtp_deny = { CODE = 5 }
			}
		}
	}
}
