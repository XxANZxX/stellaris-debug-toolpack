			} }

	effect = {

			hidden_effect = {
				from = {
					dtp_pidx_ensure = yes
					dtp_init_targets = yes
				}
				dtp_load = yes
				dtp_disarm = yes
			}
			if = {
				limit = {
					from = {
						dtp_has_perm = { P = gm }
					}
				}
				if = {
					limit = {
						exists = event_target:dtp_t
					}
					hidden_effect = {
						if = {
							limit = {
								from = {
									check_variable = { which = dtp_ry value >= 1 }
								}
							}
							event_target:dtp_t = {
								set_variable = { which = dtp_ryd value = from.dtp_ry }
								multiply_variable = { which = dtp_ryd value = 360 }
								remove_country_flag = dtp_rule_np
								set_timed_country_flag = { flag = dtp_rule_np days = dtp_ryd }
								set_country_flag = dtp_rule_np_was
								export_trigger_value_to_variable = { trigger = years_passed variable = dtp_np_endy }
								change_variable = { which = dtp_np_endy value = 2200 }
								change_variable = { which = dtp_np_endy value = from.dtp_ry }

							}

								limit = {
									is_ai = no
									dtp_listed_country = yes
								}
								country_event = { id = dtp.1211 }
							}
						}
					}
					dtp_result = { CODE = 58 }
				}
				else = {
					dtp_fail = { CODE = 58 }
				}
