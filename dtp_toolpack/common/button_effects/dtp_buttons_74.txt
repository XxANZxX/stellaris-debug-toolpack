								remove_building = building_fe_fortress
							}
						}
						else = {
							add_building = building_fe_fortress
						}
					}
					dtp_result = { CODE = 43 }
				}
				else = {
					dtp_fail = { CODE = 43 }
				}
			}
			else = {
				dtp_deny = { CODE = 43 }
			}
		}
	}
