		from = {
			has_country_flag = dtp_open
		}
		from = {
			check_variable = { which = dtp_tab value = 16 }
		}
		from = {
			check_variable = { which = dtp_gmpage value = 4 }
		}
	}
	allow = {
		custom_tooltip = { fail_text = dtp_need_perm_gm from = {
				dtp_has_perm = { P = gm }
			} }
	}
	effect = {
		hidden_effect = {
			hidden_effect = {
				from = {
					dtp_pidx_ensure = yes
					dtp_init_targets = yes
				}
				dtp_disarm = yes
			}
			if = {
				limit = {
					from = {
						dtp_has_perm = { P = gm }
					}
				}
				every_country = {
					limit = {
						is_country_type = default
						NOT = { is_same_value = from }
					}
					establish_communications_no_message = from
				}
				dtp_result = { CODE = 59 }
			}
			else = {
				dtp_deny = { CODE = 59 }
			}
		}
	}
}
