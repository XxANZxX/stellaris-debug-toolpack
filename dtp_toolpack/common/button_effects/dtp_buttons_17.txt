				if = {
					limit = {
						exists = event_target:dtp_t
					}
					event_target:dtp_t = {
						if = {
							limit = {
								has_civic = civic_corporate_hired_guns
							}
							force_remove_civic = civic_corporate_hired_guns
						}
						else = {
							force_add_civic = civic_corporate_hired_guns
						}
					}
					dtp_result = { CODE = 40 }
				}
				else = {
					dtp_fail = { CODE = 40 }
				}
			}
			else = {
				dtp_deny = { CODE = 40 }
			}
		}
	}
}
