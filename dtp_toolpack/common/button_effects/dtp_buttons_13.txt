							limit = {

					}
					event_target:dtp_cs_owner = {
						get_galaxy_setup_value = { which = dtp_cs_base setting = crisis_strength_scale }

							count = 40

						}
						while = {
							count = 40
							set_variable = { which = dtp_cs_t value = dtp_cs_mult }
							multiply_variable = { which = dtp_cs_t value = 1 }

						}

							count = 40

dtp_b_cr_reset = {
	potential = {
		from = {
			has_country_flag = dtp_open
		}
		from = {
			check_variable = { which = dtp_tab value = 16 }
		}
		from = {
			check_variable = { which = dtp_gmpage value = 4 }
		}
		exists = event_target:dtp_cs_owner
	}
	allow = {
		custom_tooltip = { fail_text = dtp_need_perm_gm from = {
				dtp_has_perm = { P = gm }
			} }
	}
	effect = {
		hidden_effect = {
			hidden_effect = {
				from = {
					dtp_pidx_ensure = yes
					dtp_init_targets = yes
				}
				dtp_disarm = yes
			}
			if = {
				limit = {
					from = {
						dtp_has_perm = { P = gm }
					}
				}
				hidden_effect = {
					if = {
						limit = {
							NOT = { exists = event_target:dtp_cs_owner }
						}
						from = {
							save_global_event_target_as = dtp_cs_owner
							set_variable = { which = dtp_cs_mult value = 1 }
						}
					}
					from = {
						if = {
							limit = {
								NOT = { is_variable_set = dtp_cst }
							}
							set_variable = { which = dtp_cst value = 1 }
						}
					}
					event_target:dtp_cs_owner = {
						get_galaxy_setup_value = { which = dtp_cs_base setting = crisis_strength_scale }
						set_variable = { which = dtp_cs_des value = 1 }
						while = {
							count = 40
							set_variable = { which = dtp_cs_t value = dtp_cs_mult }
							multiply_variable = { which = dtp_cs_t value = 2 }
							set_variable = { which = dtp_cs_d value = dtp_cs_des }
							multiply_variable = { which = dtp_cs_d value = 1 }
							if = {
								limit = {
									check_variable = { which = dtp_cs_t value <= dtp_cs_d }
								}
								multiply_crisis_strength = 2
								multiply_variable = { which = dtp_cs_mult value = 2 }
							}
						}
						while = {
							count = 40
							set_variable = { which = dtp_cs_t value = dtp_cs_mult }
							multiply_variable = { which = dtp_cs_t value = 1 }
							set_variable = { which = dtp_cs_d value = dtp_cs_des }
							multiply_variable = { which = dtp_cs_d value = 2 }
							if = {
								limit = {
									check_variable = { which = dtp_cs_t value >= dtp_cs_d }
								}
								multiply_crisis_strength = 0.5
								multiply_variable = { which = dtp_cs_mult value = 0.5 }
							}
						}
						while = {
							count = 40
							set_variable = { which = dtp_cs_t value = dtp_cs_mult }
							multiply_variable = { which = dtp_cs_t value = 1.25 }
							set_variable = { which = dtp_cs_d value = dtp_cs_des }
							multiply_variable = { which = dtp_cs_d value = 1.1 }
							if = {
								limit = {
									check_variable = { which = dtp_cs_t value <= dtp_cs_d }
								}
								multiply_crisis_strength = 1.25
								multiply_variable = { which = dtp_cs_mult value = 1.25 }
							}
						}
						while = {
							count = 40
							set_variable = { which = dtp_cs_t value = dtp_cs_mult }
							multiply_variable = { which = dtp_cs_t value = 1 }
							set_variable = { which = dtp_cs_d value = dtp_cs_des }
							multiply_variable = { which = dtp_cs_d value = 1.125 }
							if = {
								limit = {
									check_variable = { which = dtp_cs_t value >= dtp_cs_d }
								}
								multiply_crisis_strength = 0.8
								multiply_variable = { which = dtp_cs_mult value = 0.8 }
							}
						}
						set_variable = { which = dtp_cs_eff value = dtp_cs_base }
						multiply_variable = { which = dtp_cs_eff value = dtp_cs_mult }
					}
					log = "DTP CRISIS: strength set"
				}
				dtp_result = { CODE = 61 }
			}
			else = {
				dtp_deny = { CODE = 61 }
			}
		}
	}
}
