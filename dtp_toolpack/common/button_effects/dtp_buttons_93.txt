							event_target:dtp_p = {
								if = {
									limit = {
										has_modifier = angered_populace
									}
									remove_modifier = angered_populace
								}
								else = {
									if = {
										limit = {
											from = {
												check_variable = { which = dtp_years value <= 0 }
											}
										}
										add_modifier = { modifier = angered_populace days = -1 }
									}
									else = {
										set_variable = { which = dtp_ty value = from.dtp_years }
										add_modifier = { modifier = angered_populace days = 360 time_multiplier = dtp_ty }
									}
								}
							}
						}
						else_if = {
							limit = {
								from = {
									has_country_flag = dtp_pe_open
									check_variable = { which = dtp_petab value = 12 }
									check_variable = { which = dtp_pecat value = 3 }
									check_variable = { which = dtp_pepg value = 6 }
								}
							}
							event_target:dtp_p = {
								if = {
									limit = {
										has_modifier = feral_overload_01
									}
									remove_modifier = feral_overload_01
								}
								else = {
									if = {
										limit = {
											from = {
												check_variable = { which = dtp_years value <= 0 }
											}
										}
										add_modifier = { modifier = feral_overload_01 days = -1 }
									}
									else = {
										set_variable = { which = dtp_ty value = from.dtp_years }
										add_modifier = { modifier = feral_overload_01 days = 360 time_multiplier = dtp_ty }
									}
								}
							}
						}
					}
					dtp_result = { CODE = 76 }
				}
				else = {
					dtp_fail = { CODE = 76 }
				}
			}
			else = {
				dtp_deny = { CODE = 76 }
			}
		}
	}
