# Debug Toolpack 11.5.4: copy of the game file; dig sites are also visible to the Game Master country type (dtp_gamemaster).

default_site_visible_trigger = {
	OR = { is_country_type = default is_country_type = dtp_gamemaster }
	OR = {
		intel_level = {
			level >= low
			system = from.solar_system
		}
		AND = {
			exists = from
			from.planet = { is_within_borders_of = root }
		}
	}
}
