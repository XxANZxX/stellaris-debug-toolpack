				save_global_event_target_as = dtp_log1_actor
				set_variable = { which = dtp_logcode_1 value = dtp_rc }
				set_variable = { which = dtp_logres_1 value = dtp_rr }
				set_variable = { which = dtp_last_code value = dtp_rc }
				set_variable = { which = dtp_last_ok value = dtp_rr }
			}
			if = {
				limit = {
					exists = event_target:dtp_t
				}
				event_target:dtp_t = {
					save_global_event_target_as = dtp_log1_tgt
				}
			}
			else = {
				from = {
					save_global_event_target_as = dtp_log1_tgt
				}
			}
			log = "DTP | code [From.dtp_rc] | result [From.dtp_rr]"
		}
